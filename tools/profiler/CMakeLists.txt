cmake_minimum_required(VERSION 3.21)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    message(FATAL_ERROR "Configure the UMM root to build the real profiler")
endif()
get_filename_component(UMM_PROFILER_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
add_library(umm-profiler-api "${UMM_PROFILER_ROOT}/src/profiler.cpp" "${UMM_PROFILER_ROOT}/src/profiler-run.cpp" "${UMM_PROFILER_ROOT}/src/profiler-candidates.cpp")
target_compile_features(umm-profiler-api PUBLIC cxx_std_17)
target_include_directories(umm-profiler-api PUBLIC
    "${UMM_PROFILER_ROOT}/include"
    "${UMM_PROFILER_ROOT}/third_party/stable-diffusion.cpp/thirdparty")
add_executable(umm-profiler main.cpp)
target_link_libraries(umm-profiler PRIVATE umm-profiler-api)

target_link_libraries(umm-profiler-api PRIVATE umm-runtime ggml vendor::hash)
target_include_directories(umm-profiler-api PRIVATE "${UMM_PROFILER_ROOT}/src" "${UMM_PROFILER_ROOT}/third_party/llama.cpp/vendor/hash/sha256")
add_executable(umm-profiler-observer-test EXCLUDE_FROM_ALL test_observer.cpp)
target_include_directories(umm-profiler-observer-test PRIVATE "${UMM_PROFILER_ROOT}/third_party/llama.cpp/ggml/src")
target_link_libraries(umm-profiler-observer-test PRIVATE ggml Threads::Threads)
add_executable(umm-profiler-candidate-test EXCLUDE_FROM_ALL test_candidates.cpp "${UMM_PROFILER_ROOT}/src/profiler-candidates.cpp")
target_compile_features(umm-profiler-candidate-test PRIVATE cxx_std_17)
target_include_directories(umm-profiler-candidate-test PRIVATE "${UMM_PROFILER_ROOT}/src" "${UMM_PROFILER_ROOT}/third_party/stable-diffusion.cpp/thirdparty" "${UMM_PROFILER_ROOT}/third_party/llama.cpp/vendor/hash/sha256")
target_link_libraries(umm-profiler-candidate-test PRIVATE ggml vendor::hash Threads::Threads)
if(GGML_RPC)
    add_executable(umm-profiler-rpc-fence-test EXCLUDE_FROM_ALL test_rpc_fence.cpp)
    target_include_directories(umm-profiler-rpc-fence-test PRIVATE "${UMM_PROFILER_ROOT}/third_party/stable-diffusion.cpp/thirdparty")
    target_link_libraries(umm-profiler-rpc-fence-test PRIVATE ggml Threads::Threads)
endif()
add_executable(umm-profiler-contract-test EXCLUDE_FROM_ALL test_graph_contract.cpp)
target_compile_features(umm-profiler-contract-test PRIVATE cxx_std_17)
target_include_directories(umm-profiler-contract-test PRIVATE "${UMM_PROFILER_ROOT}/src" "${UMM_PROFILER_ROOT}/third_party/stable-diffusion.cpp/thirdparty")
target_link_libraries(umm-profiler-contract-test PRIVATE ggml Threads::Threads vendor::hash)

# Use the restored upstream operator harness without enabling the full test suite.
add_executable(test-backend-ops EXCLUDE_FROM_ALL "${UMM_PROFILER_ROOT}/third_party/llama.cpp/tests/test-backend-ops.cpp")
target_link_libraries(test-backend-ops PRIVATE ggml Threads::Threads)
